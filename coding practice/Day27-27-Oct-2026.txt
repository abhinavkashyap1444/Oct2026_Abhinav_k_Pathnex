 Day 27 - Serverless Scaling with Lambda and DynaimoDB
  # Ansible - Deploy Serverless Lambda Function
  - name: Deploy Serverless Lambda function
    hosts: localhost
    tasks:
      - name: Create Lambda function
        command: aws lambda create-function --function-name pathnex-lambda -- runtime nodejs14.x --role arn:aws:iam::123456789012:role/lambda-role --handler index,handler --zip-file file://lambda.zip
 
 # Terraform — Create DynamoDB Table for Serverless Application
   resource "aws_dynamodb_table" "pathnex_table" {
   name           = "pathnex-table"
   hash_key       = "id"
   billing_mode   = "PAY_PER_REQUEST"
   attribute {
    name = "id"
    type = "S"
  }
 }

 # Kubernetes _deploy Serverless Using Kubeless
  apiVersion: lubeless.io/v1beta1
  kind: function
  metadata:
     name: pathnex-lambda
     namespace: default
  spec:
     runtime: nodejs14
     handler: index.handler
     function:
       module.export = function(event, context) {
         context.succed('hello world');
	};
	
 # JenkinsFile -Lambda Function Deployment Pipeline
  pipeline {
     agent any
     stages {
        stage('Build') {
	   steps {
	       echo 'Building Lambda functiom package...'
	       sh 'zip -r lambda.zip lambda/'
	      }
	    }
	    stage('Deploy to lambda') {
	       steps {
	          script {
		      sh 'aws lambda update-function-code -- function-name pathnex-lambda --zip-file fileb://lambda.zip'
              }
	     }
	    }
	   }
